Deleting the library modules no route reaches

The measured line map records, for every module, the lines executed by the forward route, by the inverse route, and by either. A module that both routes leave at zero can still be load-bearing, because a test, a driver or another library module may import it without running it, so the map alone never licenses a deletion. Each candidate was therefore offered to a static importer scan, and a module is removed only when the scan finds no importer anywhere and no package initialiser re-exports it.

How a module is cleared for deletion

All four conditions must hold, and the scan refuses with the failing condition named when any one does not.

  1. Unreached. The map shows it measured and executing zero lines on both routes.
  2. Unimported. No importer exists in nova/, tests/, apps/, the census-kept files under benchmarks/ and scripts/, or /home/ITER/mcintos/Code/imas-ambix. The scan resolves absolute imports, relative imports at every level against the importing file's own package, importlib.import_module and __import__ string arguments, and a bare sibling import against the importing file's directory and any sys.path that file inserts. A mention only counts at a token boundary, so nova.biot.geometry inside geometryparameters is not an import.
  3. Not re-exported. No package initialiser imports it, so nova.package.Name cannot be reached through a package attribute.
  4. Not owned. It is not under nova/imas, and no plan in the sprints that own this work names it or its directory. A plan's own text is read for the module path, its dotted name and its directory; ownership that comes only from a shipped plan that has already landed does not block a deletion here.

The importer relation is then closed transitively: a module imported only by modules that are themselves deletable is deleted with them, and a module imported by any module that survives is refused with that importer's name. No candidate in this population reached deletion through the transitive path, so every accepted module has zero importers outright.

scan.py — the importer scan

"""Static importer scan.

For every module the hot-path map measures as reached by neither route, decide
whether an importer remains anywhere it would break: nova/, tests/, apps/, the
census's kept scripts and benchmarks, imas-ambix, or a live run's worktree.
Package __init__ re-exports and S28-S32 plan ownership also refuse a module; the
nova importer relation is closed transitively so a module imported only by other
deletable modules can still go.

Read-only. Writes one JSON receipt.
"""

import argparse
import ast
import json
import os
import re
import subprocess
from pathlib import Path

NOVA = "nova/"


def read(p):
    try:
        return Path(p).read_text(encoding="utf-8", errors="replace")
    except Exception:
        return ""


def walk(root, skip=()):
    for dp, dn, fns in os.walk(root):
        dn[:] = [d for d in dn if d not in skip]
        for f in fns:
            if f.endswith(".py"):
                yield os.path.join(dp, f)


def dotted(rel):
    d = rel[:-3].replace(os.sep, ".")
    return d[:-9] if d.endswith(".__init__") else d


def ispkg(rel):
    return rel.endswith("__init__.py")


def resolve(owner, level, mod, pkg):
    if level == 0:
        return mod or ""
    if owner is None:
        return ""
    parts = owner.split(".")
    base = list(parts) if pkg else parts[:-1]
    drop = level - 1
    if drop:
        base = base[:-drop] if drop < len(base) + 1 else []
    s = ".".join(base)
    if mod:
        return s + "." + mod if s else mod
    return s


def targets_of(tree, owner, pkg):
    abs_targets = set()
    bare = set()
    for n in ast.walk(tree):
        if isinstance(n, ast.Import):
            for a in n.names:
                abs_targets.add(a.name)
                bare.add(a.name.split(".")[0])
        elif isinstance(n, ast.ImportFrom):
            b = resolve(owner, n.level, n.module, pkg)
            if b:
                abs_targets.add(b)
            for a in n.names:
                if a.name == "*":
                    continue
                if b:
                    abs_targets.add(b + "." + a.name)
                bare.add(a.name)
        elif isinstance(n, ast.Call):
            f = n.func
            dyn = None
            if isinstance(f, ast.Attribute) and f.attr in ("import_module", "__import__"):
                dyn = f.attr
            elif isinstance(f, ast.Name) and f.id == "__import__":
                dyn = "imp"
            if dyn:
                for a in n.args:
                    if isinstance(a, ast.Constant) and isinstance(a.value, str):
                        abs_targets.add(a.value)
    return abs_targets, bare


def make_row(rel, tag, text):
    owner = dotted(rel) if tag == "nova" else None
    try:
        tg, br = targets_of(ast.parse(text), owner, ispkg(rel))
    except SyntaxError:
        tg, br = set(), set()
    return {
        "rel": rel,
        "tag": tag,
        "text": text,
        "targets": tg,
        "bare": br,
        "pkg": ispkg(rel),
        "owner": owner,
        "lines": text.count("\n"),
    }


def scan_dir(root, tag, prefix="", skip=()):
    rows = []
    for p in walk(root, skip):
        rel = prefix + os.path.relpath(p, root).replace(os.sep, "/")
        rows.append(make_row(rel, tag, read(p)))
    return rows


def wt_changed_py(wt):
    """Python files a live worktree changed against its head or main."""
    paths = set()
    cmds = (
        ["status", "--porcelain", "-uno"],
        ["diff", "--name-only", "HEAD"],
        ["diff", "--name-only", "main"],
    )
    for args in cmds:
        try:
            out = subprocess.run(
                ["git", "-C", wt] + args, capture_output=True, text=True, timeout=120
            ).stdout
        except Exception:
            continue
        for line in out.splitlines():
            p = line.strip()
            if args[0] == "status":
                p = line[3:].strip()
                if "->" in p:
                    p = p.split("->")[-1].strip()
            if p.endswith(".py"):
                paths.add(p)
    return paths


def mentions(text, needle):
    if not needle:
        return False
    pat = r"(?<![A-Za-z0-9_.])" + re.escape(needle) + r"(?![A-Za-z0-9_])"
    return re.search(pat, text) is not None


def row_hits(row, needles):
    text = row["text"]
    targets = row["targets"]
    bare = row["bare"]
    rel = row["rel"]
    out = []
    for mod, nd in needles.items():
        if row["tag"] == "nova" and rel == mod:
            continue
        d = nd["dotted"]
        hit = d in targets
        if not hit:
            for t in targets:
                if t == d or t.startswith(d + "."):
                    hit = True
                    break
        if not hit and nd["stem"] and nd["stem"] in text:
            if mentions(text, d) or mentions(text, nd["path"]):
                hit = True
        if not hit and nd["stem"] in bare:
            if "sys.path" in text or os.path.dirname(rel) == nd["dir"]:
                hit = True
        if hit:
            out.append(mod)
    return out


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--base", required=True)
    ap.add_argument("--map", required=True)
    ap.add_argument("--census", required=True)
    ap.add_argument("--ambix", required=True)
    ap.add_argument("--plans-dir", required=True)
    ap.add_argument("--live-worktrees", required=True)
    ap.add_argument("--out", required=True)
    args = ap.parse_args()

    hmap = json.load(open(args.map))
    census = json.load(open(args.census))
    modules = hmap["modules"]

    cands = {}
    excl = {"not_measured": [], "imas": [], "reached": 0}
    for m in modules:
        mod = m["module"]
        if not mod.startswith(NOVA):
            continue
        if not m.get("measured"):
            excl["not_measured"].append(mod)
        elif mod.startswith("nova/imas/"):
            excl["imas"].append(mod)
        elif m["union"]["executed"] != 0:
            excl["reached"] += 1
        else:
            cands[mod] = m

    keep = [v["path"] for v in census["verdicts"] if v["verdict"] == "keep"]

    scanned = scan_dir(os.path.join(args.base, "nova"), "nova", prefix="nova/")
    for tag in ("tests", "apps"):
        d = os.path.join(args.base, tag)
        if os.path.isdir(d):
            scanned += scan_dir(d, tag, prefix=tag + "/")
    for kp in keep:
        if not (kp.startswith("scripts/") or kp.startswith("benchmarks/")):
            continue
        full = os.path.join(args.base, kp)
        if os.path.isfile(full):
            scanned.append(make_row(kp, "script", read(full)))
    for sd in ("imas_ambix", "src", "tests", "scripts", "skills"):
        d = os.path.join(args.ambix, sd)
        if os.path.isdir(d):
            scanned += scan_dir(
                d, "ambix",
                skip=(".venv", ".git", "node_modules",
                      "site-packages", "__pycache__", "build", "dist"),
            )

    live = []
    if os.path.isfile(args.live_worktrees):
        for line in Path(args.live_worktrees).read_text().splitlines():
            line = line.strip()
            if line and os.path.isdir(line):
                live.append(line)

    needles = {}
    for mod in cands:
        needles[mod] = {
            "dotted": dotted(mod),
            "path": mod,
            "stem": os.path.basename(mod)[:-3],
            "dir": os.path.dirname(mod),
        }

    importers = {mod: [] for mod in cands}
    for row in scanned:
        for mod in row_hits(row, needles):
            importers[mod].append(
                {"tag": row["tag"], "rel": row["rel"], "owner": row["owner"], "pkg": row["pkg"]}
            )
    for wt in live:
        for rel in sorted(wt_changed_py(wt)):
            p = os.path.join(wt, rel)
            if not os.path.isfile(p):
                continue
            tag = rel.split("/", 1)[0]
            row = make_row(rel, tag if tag in ("nova", "tests", "apps") else "nova", read(p))
            for mod in row_hits(row, needles):
                importers[mod].append(
                    {"tag": "worktree", "rel": rel, "wt": wt,
                     "owner": row["owner"], "pkg": row["pkg"]}
                )

    plans = {}
    for pf in sorted(Path(args.plans_dir).glob("*.html")):
        t = read(str(pf))
        sp = re.search(r'name="plan-sprint" content="([^"]{0,8})"', t)
        sprint = sp.group(1) if sp else ""
        if sprint in ("S28", "S29", "S30", "S31", "S32"):
            plans[pf.name] = re.sub(r"<[^>]+>", " ", t)

    def plan_names(mod):
        d = needles[mod]["dotted"]
        names = []
        for pn, body in plans.items():
            if mod in body or (d and d in body):
                names.append([pn, "module"])
            elif needles[mod]["dir"] in body:
                names.append([pn, "directory"])
        return names

    lines = {r["rel"]: r["lines"] for r in scanned if r["tag"] == "nova"}
    module_by_path = {m["module"]: m for m in modules}

    def decide(mod, imps):
        """(verdict, reason) for one module path, independent of the fixed point."""
        if mod in module_by_path and mod not in cands:
            m = module_by_path[mod]
            if not m.get("measured"):
                return ("refuse", "not measured by either route")
            if mod.startswith("nova/imas/"):
                return ("refuse", "under nova/imas (S30, out of scope)")
            return ("refuse", "reached by a route (union %d lines)" % m["union"]["executed"])
        if mod not in module_by_path:
            return ("refuse", "not under nova/")
        if ispkg(mod):
            return ("refuse", "package initialiser")
        pn = plan_names(mod)
        if pn:
            return ("refuse", "named by S28-S32 plan %s (%s)" % (pn[0][0], pn[0][1]))
        for imp in imps:
            if imp["tag"] in ("tests", "apps", "script", "ambix"):
                return ("refuse", "imported by %s %s" % (imp["tag"], imp["rel"]))
            if imp["tag"] == "worktree":
                return ("refuse", "imported by live worktree %s" % imp.get("wt"))
        return ("transitive", "")

    status = {mod: decide(mod, importers[mod]) for mod in cands}
    deletable = {mod for mod, v in status.items() if v[0] == "transitive"}
    changed = True
    while changed:
        changed = False
        for mod in list(deletable):
            for imp in importers[mod]:
                if imp["tag"] != "nova":
                    continue
                if imp["pkg"]:
                    status[mod] = ("refuse", "re-exported by package __init__ %s" % imp["rel"])
                    deletable.discard(mod)
                    changed = True
                    break
                owner = imp["owner"]
                if owner == mod:
                    continue
                if owner not in deletable:
                    status[mod] = ("refuse", "imported by kept nova module %s" % owner)
                    deletable.discard(mod)
                    changed = True
                    break

    per_pkg = {}
    per_dir = {}
    for mod in sorted(deletable):
        parts = mod.split("/")
        pkg = "/".join(parts[:2]) if len(parts) > 1 else mod
        per_pkg[pkg] = per_pkg.get(pkg, 0) + (lines.get(mod) or 0)
        per_dir[mod] = lines.get(mod) or 0

    # Negative control 1: a reached module offered for deletion is refused.
    reached_mod = None
    for m in modules:
        mm = m["module"]
        if (mm.startswith("nova/") and not mm.startswith("nova/imas/")
                and not mm.endswith("__init__.py")
                and m.get("measured") and m["union"]["executed"] != 0):
            reached_mod = mm
            break

    # Negative control 2: an unreached module whose only importer is imas-ambix.
    ambix_only = [m for m in cands if {i["tag"] for i in importers[m]} == {"ambix"}]
    if ambix_only:
        nc2_mod = ambix_only[0]
        nc2_imps = importers[nc2_mod]
        nc2_note = "natural: only importer is imas-ambix"
    else:
        nc2_mod = sorted(deletable)[0] if deletable else None
        nc2_imps = (importers[nc2_mod] if nc2_mod else []) + [
            {"tag": "ambix", "rel": "imas_ambix/live_import_probe.py", "owner": None, "pkg": False}
        ]
        nc2_note = "synthetic: ambix importer injected"

    rec = {}
    for mod in cands:
        v, reason = status[mod]
        rec[mod] = {
            "module": mod,
            "lines": lines.get(mod),
            "verdict": v,
            "deletable": mod in deletable,
            "reason": reason,
            "importers": importers[mod],
            "plan": plan_names(mod),
        }

    out = {
        "revision": hmap.get("revision"),
        "counts": {
            "candidates": len(cands),
            "deletable": len(deletable),
            "refused": len(cands) - len(deletable),
            "reached_excluded": excl["reached"],
            "imas_excluded": len(excl["imas"]),
            "not_measured": len(excl["not_measured"]),
        },
        "deletable": sorted(deletable),
        "deletable_lines": sum(per_dir.values()),
        "lines_per_package": per_pkg,
        "candidates": rec,
        "negative_controls": {
            "reached_offered": {
                "module": reached_mod,
                "verdict": decide(reached_mod, []) if reached_mod else None,
            },
            "ambix_only": {
                "module": nc2_mod,
                "note": nc2_note,
                "verdict": decide(nc2_mod, nc2_imps) if nc2_mod else None,
            },
        },
        "live_worktrees": live,
        "plan_sprints": sorted(plans),

delete.py — the deletion driver

"""Apply the importer-scan verdicts: offer (classify) or delete (git rm).

--offer PATH [PATH...]   print ACCEPT/REFUSE and the reason for each offered
                         module as the scan's check (1) reads it. Used for the
                         negative controls; deletes nothing.
--apply                  git-rm every module the scan marks deletable, minus
                         the hold-backs, refusing if the tree is not clean.

A module is deletable only when the scan verdict is `transitive` AND it is not
held back. A held-back module is named with its reason and left in place. The
refusal is the check the plan's done-when asks for in (1).
"""

import argparse
import json
import subprocess
import sys

HOLD = {
    "nova/utilities/ANSYS.py": (
        "named by the landed measured-line-map driver "
        "scripts/prototypes/hot_path_map/job.sh as its negative-control module; "
        "map.py refuses to write the map when that module carries no coverage "
        "entry, so deleting it makes the landed 1 record irreproducible"
    ),
}


def classify(mod, scan, bypath):
    rec = scan["candidates"].get(mod)
    if rec is None:
        m = bypath.get(mod)
        if m is not None and m.get("measured"):
            if mod.startswith("nova/imas/"):
                return "REFUSE", "under nova/imas (S30, out of scope)"
            return "REFUSE", "reached by a route (union %d lines)" % m["union"]["executed"]
        return "REFUSE", "not a module under nova/ in the hot-path map"
    if mod in HOLD:
        return "REFUSE", "held back: " + HOLD[mod]
    if not rec["deletable"]:
        return "REFUSE", rec["reason"]
    return "ACCEPT", "unreached and unimported (%d lines)" % (rec["lines"] or 0)


def git(base, *a):
    return subprocess.run(["git", "-C", base] + list(a), capture_output=True, text=True)


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--base", required=True)
    ap.add_argument("--scan", required=True)
    ap.add_argument("--map", required=True)
    ap.add_argument("--offer", nargs="*")
    ap.add_argument("--apply", action="store_true")
    args = ap.parse_args()

    scan = json.load(open(args.scan))
    bypath = {m["module"]: m for m in json.load(open(args.map))["modules"]}

    if args.offer is not None:
        for mod in args.offer:
            verdict, why = classify(mod, scan, bypath)
            print("%s %s -- %s" % (verdict, mod, why))
        return

    if not args.apply:
        ap.error("one of --offer or --apply is required")

    st = git(args.base, "status", "--porcelain", "-uno")
    if st.stdout.strip():
        print("REFUSE: tracked tree is dirty:\n" + st.stdout, file=sys.stderr)
        sys.exit(3)

    targets = []
    held = []
    for mod in sorted(scan["deletable"]):
        verdict, why = classify(mod, scan, bypath)
        if verdict == "ACCEPT":
            targets.append(mod)
        else:
            held.append((mod, why))

    pkg = {}
    for mod in targets:
        lines = scan["candidates"][mod]["lines"] or 0
        p = "/".join(mod.split("/")[:2])
        pkg[p] = pkg.get(p, 0) + lines

    print("deleting %d modules" % len(targets))
    res = git(args.base, "rm", "--", *targets)
    sys.stdout.write(res.stdout)
    sys.stderr.write(res.stderr)
    if res.returncode != 0:
        sys.exit(res.returncode)

    for mod, why in held:
        print("HELD %s -- %s" % (mod, why))
    print("deleted_lines", sum(pkg.values()))
    print("lines_per_package", json.dumps(pkg, sort_keys=True))
    for mod in targets:
        import os
        if os.path.exists(os.path.join(args.base, mod)):
            print("STILL PRESENT:", mod, file=sys.stderr)
    print("absent_check", "all deleted paths absent")


if __name__ == "__main__":
    main()
